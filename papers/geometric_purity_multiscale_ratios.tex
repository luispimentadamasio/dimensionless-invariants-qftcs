\begin{abstract}
To analyze physical systems spanning over one hundred orders of magnitude without losing structural clarity in raw dimensional units, we construct four dimensionless diagnostic invariants: \textbf{Weyl Conformal Purity} ($\chi$), the \textbf{Gravity-to-Thermal Decoherence Crossover Ratio} ($\mathcal{R}_{\mathrm{DP}}$), \textbf{Penrose--Zel'dovich Rotational Extraction Gain} ($\mathcal{E}_{\mathrm{Penrose}}$), and the \textbf{Semiclassical Breakdown Index} ($\Xi_{\mathrm{QG}}$). Derived from the Seeley--DeWitt $a_4$ trace anomaly, Buckingham-$\Pi$ dimensional analysis, Novikov--Thorne binding energies, and the $L_\infty$ Chebyshev supremum norm, these metrics separate geometric vacuum polarization from environmental noise while establishing strict epistemic boundaries for semiclassical gravity.
\end{abstract}

\noindent\textbf{Keywords:} Weyl trace anomaly, Penrose--Diósi collapse, Buckingham-$\Pi$ theorem, Novikov--Thorne efficiency, Semiclassical breakdown, Dimensionless invariants

\section{First-Principles Mathematical Derivations}

\subsection{Weyl Conformal Purity ($\chi$) via Seeley--DeWitt Trace Anomaly}
In classical massless field theory, the stress-energy tensor is traceless ($T^\mu_{\;\mu} = 0$) due to invariance under local Weyl rescalings ($g_{\mu\nu} \to \Omega^2(x) g_{\mu\nu}$). In curved spacetime, quantum vacuum fluctuations break this symmetry, generating the \textbf{Weyl Conformal Trace Anomaly} ($\langle \hat{T}^\mu_{\;\mu} \rangle_{\mathrm{ren}}$) governed by the one-loop Seeley--DeWitt $a_4$ heat-kernel coefficient:
\begin{equation}
\mathcal{A}_W \equiv \frac{\langle \hat{T}^\mu_{\;\mu} \rangle_{\mathrm{ren}}}{\rho_{\mathrm{vac,ref}}} = \frac{\mathcal{C}_{\mathrm{clamped}}^2}{60\pi^2}
\end{equation}
where $\mathcal{C}_{\mathrm{clamped}} = \min(\mathcal{C}_{\mathrm{hor}}, \mathcal{C}_{\mathrm{eff}})$ bounds the dimensionless compactness at the Kerr--Newman horizon $\mathcal{C}_{\mathrm{hor}} = 2 / (1 + \sqrt{1 - a_*^2 - Q_*^2})$. Normalizing $\mathcal{A}_W \ge 0$ by the characteristic one-loop phase-space factor ($60$) yields the bounded \textbf{Weyl Conformal Purity}:
\begin{equation}
\chi \equiv \frac{1}{1 + 60\,\mathcal{A}_W} = \frac{1}{1 + \mathcal{C}_{\mathrm{clamped}}^2 / \pi^2} \in (0, 1]
\end{equation}
\begin{itemize}
    \item \textbf{Minkowski Limit ($\mathcal{C}_{\mathrm{clamped}} = 0$):} $\mathcal{A}_W = 0 \implies \chi = 1.0000$.
    \item \textbf{Schwarzschild Horizon ($\mathcal{C}_{\mathrm{clamped}} = 1$):} $\chi = \pi^2 / (\pi^2 + 1) \approx 0.9080$.
    \item \textbf{Extremal Kerr--Newman Horizon ($\mathcal{C}_{\mathrm{clamped}} \to 2$):} $\chi = \pi^2 / (\pi^2 + 4) \approx 0.7116$.
\end{itemize}

\subsection{Gravity-to-Thermal Crossover Ratio ($\mathcal{R}_{\mathrm{DP}}$) via Buckingham-$\Pi$ Scaling}
Comparing the Penrose--Diósi gravitational self-energy collapse rate $\Gamma_{\mathrm{grav}} = (M_{\mathrm{eff}} / M_P)^2 (c / L)$ against the Caldeira--Leggett thermal photon scattering rate $\Gamma_{\mathrm{th}} = L^2 (k_B T_{\mathrm{eff}})^3 / (\hbar^3 c^2)$ eliminates time units completely, forming a dimensionless Buckingham-$\Pi$ group:
\begin{equation}
\mathcal{R}_{\mathrm{DP}} \equiv \frac{\Gamma_{\mathrm{grav}}}{\Gamma_{\mathrm{th}}} = \left(\frac{M_{\mathrm{eff}}}{M_P}\right)^2 \frac{\hbar^2 c^3}{L^3 (k_B T_{\mathrm{eff}})^3} = \left(\frac{M_{\mathrm{eff}}}{M_P}\right)^2 \left(\frac{\bar{\lambda}_{\mathrm{th}}}{L}\right)^3
\end{equation}
where $\bar{\lambda}_{\mathrm{th}} = \hbar c / (k_B T_{\mathrm{eff}})$ is the reduced thermal wavelength. The critical crossover $\mathcal{R}_{\mathrm{DP}} = 1$ occurs along the mesoscopic mass-temperature boundary $M_{\mathrm{eff}} / M_P = (L / \bar{\lambda}_{\mathrm{th}})^{3/2}$.

\subsection{Penrose--Zel'dovich Rotational Extraction Gain ($\mathcal{E}_{\mathrm{Penrose}}$)}
To quantify rotational energy extraction from a spinning Kerr--Newman ergosphere at polar angle $\theta$, we combine three dimensionless physical factors:
\begin{enumerate}
    \item \textbf{Zel'dovich--Starobinsky Superradiant Wave Gain:} $\mathcal{G}_{\mathrm{SR}} = \max(0, 1 - f_{\mathrm{local}} / f_H)$, where $f_H = c a_* / (4\pi r_+)$ is the horizon angular frequency.
    \item \textbf{Polar Ergosphere Coupling:} $\sin^2\theta$, reflecting the vanishing of ergosphere thickness along the rotation axis ($\theta = 0^\circ$) and peak frame-dragging at the equator ($\theta = 90^\circ$).
    \item \textbf{Novikov--Thorne Radiative Binding Efficiency:} $\eta_{\mathrm{NT}}(r_{\mathrm{ISCO}}) = 1 - \sqrt{1 - 2r_g / (3 r_{\mathrm{ISCO}})}$, ranging from $5.72\%$ (Schwarzschild) to $42.26\%$ (extremal Kerr).
\end{enumerate}
Taking their product defines the \textbf{Penrose--Zel'dovich Extraction Gain}:
\begin{equation}
\mathcal{E}_{\mathrm{Penrose}}(\theta) \equiv \max\left(0,\; 1 - \frac{f_{\mathrm{local}}}{f_H}\right) \cdot \sin^2\theta \cdot \eta_{\mathrm{NT}}(r_{\mathrm{ISCO}})
\end{equation}

\subsection{Semiclassical Breakdown Index ($\Xi_{\mathrm{QG}}$) via $L_\infty$ Supremum Norm}
Semiclassical Einstein gravity ($G_{\mu\nu} = \frac{8\pi G}{c^4} \langle \hat{T}_{\mu\nu} \rangle_{\mathrm{ren}}$) holds only while four independent dimensionless ratios remain strictly below unity:
\begin{itemize}
    \item \textbf{Minimal Planck Length Cutoff:} $\Pi_1 = \ell_P / L$
    \item \textbf{Stretched-Horizon Membrane Cutoff:} $\Pi_2 = 10^{-6} / N^2(\theta)$
    \item \textbf{Sub-Extremal Cosmic Censorship Cutoff:} $\Pi_3 = (a_*^2 + Q_*^2) / 0.9998$
    \item \textbf{Planck Thermal Foam Ceiling:} $\Pi_4 = T_{\mathrm{eff}} / T_P$
\end{itemize}
Taking the $L_\infty$ (Chebyshev) supremum norm defines the \textbf{Semiclassical Breakdown Index}:

\Xi_{\mathrm{QG}} \equiv \big\|(\Pi_1, \Pi_2, \Pi_3, \Pi_4)\big\|_\infty = \max\left(\frac{\ell_P}{L},\; \frac{10^{-6}}{N^2(\theta)},\; \frac{a_*^2 + Q_*^2}{0.9998},\; \frac{T_{\mathrm{eff}}}{T_P}\right)

\section{Plain-Language Physical \& Operational Guide}

\begin{itemize}

    \item \textbf{Gravity-to-Thermal Crossover ($\mathcal{R}_{\mathrm{DP}}$):} Acts as a dimensionless tug-of-war gauge between environmental thermal decoherence ($\mathcal{R}_{\mathrm{DP}} < 1$) and Penrose--Diósi gravitational self-energy collapse ($\mathcal{R}_{\mathrm{DP}} > 1$), with $\mathcal{R}_{\mathrm{DP}} = 1$ marking the exact quantum-to-classical mesoscopic threshold.

\end{itemize}
