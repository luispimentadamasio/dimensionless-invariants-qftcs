\begin{abstract}
We present a self-contained semiclassical formulation of unitary black hole evaporation that couples dynamical horizon collapse and Quantum Extremal Surface (QES) entanglement-wedge reconstruction with a unified set of dimensionless diagnostic invariants. Modeling exponential mode peeling across an ingoing Vaidya horizon, we trace the thermodynamic transition from early-time linear entanglement growth to late-time radiation purification via the Euclidean replica-wormhole saddle exchange. To audit microstate unitarity, horizon blueshift regularization, and holographic information bounds continuously along Kerr--Newman evaporation trajectories, we integrate the Zero-Angular-Momentum Observer (ZAMO) dilation factor ($\rho_\tau$), Weyl conformal purity ($\chi$), decoherence action ($\Omega$), holographic coherence floor ($\Phi_{\text{floor}}$), timeline integrity ($\Phi_\tau$), surviving coherent cycles ($\mathcal{N}_{\text{eff}}$), the Lloyd--Bekenstein gate ratio ($\eta_{\text{Lloyd}}$), and the $L_\infty$ semiclassical breakdown index ($\Xi_{\text{QG}}$).
\end{abstract}

\noindent\textbf{Keywords:} Black Hole Information Paradox, Quantum Extremal Surfaces, Island Formula, Replica Wormholes, Page Curve, Unitary Evaporation, Dimensionless Invariants

\section{Dynamical Horizon Collapse and Kinematical Redshift}

Consider a spherically symmetric collapsing star modeled by the ingoing \textbf{Vaidya spacetime metric}:
\begin{equation}
ds^2 = -\left(1 - \frac{2GM(v)}{c^2 r}\right) c^2 dv^2 + 2c\, dv\, dr + r^2 d\Omega^2
\end{equation}
where $v$ denotes advanced null time and $M(v)$ is the evolving mass profile.

\begin{itemize}
\item \textbf{Exponential Mode Peeling and Bogoliubov Coefficients:} Tracing outgoing null rays backward from future null infinity ($\mathcal{I}^+$) through the collapsing center to past null infinity ($\mathcal{I}^-$) relates asymptotic retarded time $u$ to the affine horizon coordinate $U$ via $U = -\kappa^{-1} e^{-\kappa u}$, where $\kappa$ is the horizon surface gravity. Evaluating the Klein--Gordon inner product between early-time ($\mathcal{I}^-$) and late-time ($\mathcal{I}^+$) field modes yields Bogoliubov coefficients $\alpha_{\omega\omega'}$ and $\beta_{\omega\omega'}$ satisfying a thermal Planck spectrum at the \textbf{Hawking temperature} \cite{Hawking1975}:
\begin{equation}
T_H = \frac{\hbar \kappa}{2\pi k_B c}
\end{equation}

\item \textbf{ZAMO Dilation Factor ($\rho_\tau$):} Relates proper time intervals along the local Zero-Angular-Momentum Observer frame ($d\tau_{\text{pre}}$) to distant observer proper time ($d\tau_{\text{obs}}$) via $\rho_\tau = d\tau_{\text{obs}} / d\tau_{\text{pre}}$ \cite{ADM1962,BPT1972}. It transforms local horizon-frame temperature $T_{\text{eff}}$ into the Tolman--Ehrenfest redshifted temperature measured at infinity, $T_{\text{obs}} = T_{\text{eff}} / \rho_\tau$. Under the stretched-horizon membrane regulator ($N^2 \ge 10^{-6}$) \cite{TPM1986}, $\rho_\tau$ remains strictly bounded ($\rho_\tau \le 10^3$), preventing unphysical infinite-blueshift divergences near the horizon.

\item \textbf{Weyl Conformal Purity ($\chi$):} Quantifies static geometric vacuum polarization induced by background tidal curvature ($\mathcal{A}_W = \mathcal{C}_{\text{clamped}}^2 / 60\pi^2$) derived from the Seeley--DeWitt $a_4$ trace anomaly \cite{DeWitt1965}:
\begin{equation}
\chi = \frac{1}{1 + 60\,\mathcal{A}_W} = \frac{1}{1 + \mathcal{C}_{\text{clamped}}^2 / \pi^2}
\end{equation}
In flat Minkowski space ($\mathcal{C}_{\text{clamped}} = 0$), $\chi = 1$; at a horizon, $\chi$ drops below unity ($\approx 0.908$ for Schwarzschild, $\approx 0.712$ for extremal Kerr--Newman), isolating time-independent curvature mode mixing from thermal particle emission.
\end{itemize}

\section{Generalized Entropy and the Quantum Extremal Surface}

In semiclassical gravity, the von Neumann entropy of Hawking radiation ($S_{\text{rad}}$) collected in a distant region $R$ diverges near the horizon due to short-distance ultraviolet entanglement across the boundary. Regulating this divergence requires pairing geometric horizon area with renormalized bulk field entropy on a spacelike slice $\Sigma_X$.

\begin{itemize}
\item \textbf{Generalized Entropy Functional ($S_{\text{gen}}$):} Allowing the radiation's entanglement wedge to include a disconnected interior \textbf{island} $I$ bounded by a codimension-2 surface $\partial I = X$, the generalized entropy is:
\begin{equation}
S_{\text{gen}}(X) = \frac{\text{Area}(X)}{4 G \hbar} + S_{\text{bulk}}(R \cup I)
\end{equation}

\item \textbf{Extremization and Minimization (The Island Rule):} Varying $X$ along orthogonal future-directed null normals ($k^\mu$ outward, $l^\mu$ inward) until the quantum expansions vanish ($\Theta_k = \Theta_l = 0$) and selecting the global minimum over all extremal saddles gives \cite{AEMM2019,Penington2020}:
\begin{equation}
S_{\text{rad}}(R) = \min_I \left\{ \operatorname{ext}_{\partial I} \left[ \frac{\text{Area}(\partial I)}{4 G \hbar} + S_{\text{bulk}}(R \cup I) \right] \right\}
\end{equation}

\item \textbf{Euclidean Replica Wormhole Saddle Exchange:} Evaluating the $n$-sheeted gravitational path integral via the \textbf{Replica Trick}, $S_{\text{rad}} = -\lim_{n \to 1} \frac{\partial}{\partial n} \ln (Z_n / Z_1^n)$, exposes two competing topological saddles:
\begin{enumerate}
\item \textbf{The Hawking Saddle ($I = \emptyset$, Disconnected Replicas):} Produces linear entanglement growth $S_{\text{rad}}^{(\text{Hawking})}(t) \approx \frac{c_{\text{eff}}}{12} \kappa t$, leading to apparent information loss \cite{Hawking1975,Page1993}.
\item \textbf{The Replica Wormhole Saddle ($I \neq \emptyset$, Connected Replicas):} Connects the $n$ copies through a wormhole throat branched at $\partial I$, yielding $S_{\text{rad}}^{(\text{Island})}(t) \approx \frac{\text{Area}(r_+(t))}{4 G \hbar} = S_{\text{BH}}(t)$ \cite{AEMM2019,Penington2020}.
\end{enumerate}
\end{itemize}

\section{Unitary Page Curve Evolution and Microscopic Coherence Bounds}

Summing over both saddles in the Euclidean path integral forces the smaller Euclidean action to dominate exponentially, yielding the unitary \textbf{Page Curve} \cite{Page1993}:
\begin{equation}
S_{\text{rad}}(t) = \min\big[S_{\text{BH,init}} - S_{\text{BH}}(t),\; S_{\text{BH}}(t)\big]
\end{equation}
\begin{itemize}
\item \textbf{Pre-Page Regime ($t < t_{\text{Page}}$):} The trivial saddle ($I = \emptyset$) dominates, and $S_{\text{rad}}(t)$ rises as outgoing quanta entangle with interior partners.
\item \textbf{Page Time ($t = t_{\text{Page}}$):} When the horizon area shrinks to half its initial value, $S_{\text{rad}}^{(\text{Hawking})}(t_{\text{Page}}) = S_{\text{BH}}(t_{\text{Page}})$, triggering the phase transition to the non-trivial Quantum Extremal Surface.
\item \textbf{Post-Page Regime ($t > t_{\text{Page}}$):} Late Hawking quanta purify early radiation, and $S_{\text{rad}}(t)$ tracks $S_{\text{BH}}(t)$ downward along the Kerr--Newman evaporation trajectory:
\begin{equation}
\frac{dM}{dt} = -\alpha_H \frac{f_{\text{KN}}^2}{M^2}, \qquad f_{\text{KN}} = \frac{2\chi_{\text{ext}}}{1 + \chi_{\text{ext}}}, \qquad \chi_{\text{ext}} = \sqrt{1 - a_*^2 - Q_*^2}
\end{equation}
\end{itemize}

\subsection{How Custom Invariants Enforce and Audit Microscopic Unitarity}

\begin{itemize}
\item \textbf{Decoherence Action ($\Omega$) and Timeline Integrity ($\Phi_\tau$):} Along the evaporating worldline, thermal Hawking scattering ($\Gamma_{\text{th}}$) and gravitational self-energy dephasing ($\Gamma_{\text{grav}}$) accumulate dimensionless decoherence action $\Omega = (\Gamma_{\text{th}} + \Gamma_{\text{grav}})\,t_{\text{pre}}$ \cite{GKS1976,Lindblad1976}. Rather than allowing quantum state purity to vanish identically ($\operatorname{Tr}(\rho^2) \to 0$), \textbf{Timeline Integrity ($\Phi_\tau$)} bounds coherence decay from below:
\begin{equation}
\Phi_\tau(\Omega) = \Phi_{\text{floor}} + (1 - \Phi_{\text{floor}})\,e^{-\Omega}
\end{equation}

\item \textbf{Holographic Coherence Floor ($\Phi_{\text{floor}}$):} Derived from the Bekenstein--Hawking Hilbert-space dimension $N_{\text{dof}} = 1 + \pi \cdot 10^{2\zeta}$ (where $\zeta = \log_{10}(r_+ / \ell_P)$), $\Phi_{\text{floor}} = (1 + \pi \cdot 10^{2\zeta})^{-1}$ represents the irreducible purity of a maximally scrambled state inside a causal sphere of radius $r_+$. As evaporation shrinks the horizon toward the Planck length ($\zeta \to 0$), $\Phi_{\text{floor}}$ rises smoothly from $\approx 10^{-70}$ to $(1 + \pi)^{-1} \approx 0.2414$, preserving residual microstate correlations.

\item \textbf{Surviving Coherent Cycles ($\mathcal{N}_{\text{eff}}$) and Lloyd--Bekenstein Ratio ($\eta_{\text{Lloyd}}$):} Weighting total emitted wave oscillations ($\mathcal{N}_\phi = f_{\text{local}}\,t_{\text{pre}}$) by $\Phi_\tau$ yields $\mathcal{N}_{\text{eff}} = \mathcal{N}_\phi \cdot \Phi_\tau$. Comparing $\mathcal{N}_{\text{eff}}$ against the shrinking horizon's holographic entropy capacity defines the \textbf{Lloyd--Bekenstein Gate Ratio} \cite{ML1998,Lloyd2000}:
\begin{equation}
\eta_{\text{Lloyd}} = \frac{\mathcal{N}_{\text{eff}}}{S_{\text{BH}} / k_B} \le 1
\end{equation}

\item \textbf{Semiclassical Breakdown Index ($\Xi_{\text{QG}}$):} As $M(t) \to M_P$ and $r_+ \to \ell_P$, the minimal-length ratio $\Pi_1 = \ell_P / L$ and thermal-foam ratio $\Pi_4 = T_H / T_P$ approach unity. The $L_\infty$ supremum monitor $\Xi_{\text{QG}} = \max(\Pi_1, \Pi_2, \Pi_3, \Pi_4) \to 1$ signals the exact breakdown of smooth semiclassical geometry, halting evaporation at a stable \textbf{Planck-mass relic ($M_P$)}.
\end{itemize}
